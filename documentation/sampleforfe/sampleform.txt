import { Component, OnInit } from '@angular/core';
import { FormBuilder, FormGroup } from '@angular/forms';
import { ActivatedRoute, Router } from '@angular/router';

@Component({
  selector: 'app-project-form',
  templateUrl: './project-form.component.html',
  styleUrls: ['./project-form.component.scss']
})
export class ProjectFormComponent implements OnInit {
  projectForm: FormGroup;
  isNewProject = false;
  projectTypes = ['aaa', 'bbb', 'ccc']; // Example types
  displayedColumns: string[] = ['name', 'type'];
  assets = [
    { name: 'Asset 1', type: 'eee' },
    { name: 'Asset 2', type: 'fff' }
  ]; // Mocked asset data

  constructor(
    private fb: FormBuilder,
    private route: ActivatedRoute,
    private router: Router
  ) {
    this.projectForm = this.fb.group({
      name: [''],
      startDate: [''],
      endDate: [''],
      type: [''],
      notes: ['']
    });
  }

  ngOnInit() {
    const projectId = this.route.snapshot.paramMap.get('id');
    this.isNewProject = !projectId; // If no ID, it's a new project

    if (!this.isNewProject) {
      // Load project details for editing (replace with API call)
      const projectData = { name: 'Project A', startDate: '2025-01-01', endDate: '2025-12-31', type: 'aaa', notes: 'Some notes' };
      this.projectForm.patchValue(projectData);
    }
  }

  saveProject() {
    if (this.projectForm.valid) {
      const projectData = this.projectForm.value;
      console.log('Saved Project:', projectData);
      this.router.navigate(['/']); // Redirect back to project list
    }
  }
}
