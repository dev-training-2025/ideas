public class DatabaseSettings
{
    public string ConnectionString { get; set; } = null!;
    public string DatabaseName { get; set; } = null!;
}


MyService.cs:


using MongoDB.Driver;

public class MyService
{
    private readonly IMongoDatabase _database;

    public MyService(DatabaseSettings settings)
    {
        var client = new MongoClient(settings.ConnectionString);
        _database = client.GetDatabase(settings.DatabaseName);
    }

    public IMongoCollection<T> GetCollection<T>(string collectionName)
    {
        return _database.GetCollection<T>(collectionName);
    }
}



Step 3: Register Services in Program.cs

var builder = WebApplication.CreateBuilder(args);

builder.Services.Configure<DatabaseSettings>(builder.Configuration.GetSection("DatabaseSettings"));
builder.Services.AddSingleton<MyService>();

builder.Services.AddControllers();

var app = builder.Build();

app.UseAuthorization();
app.MapControllers();

app.Run();



Step 4: Define Controllers
Create a controller to handle API requests: MyController.cs:

using Microsoft.AspNetCore.Mvc;
using MongoDB.Driver;

[ApiController]
[Route("api/[controller]")]
public class MyController : ControllerBase
{
    private readonly MyService _service;

    public MyController(MyService service)
    {
        _service = service;
    }

    [HttpGet("{collection}")]
    public async Task<IActionResult> Get(string collection)
    {
        var data = await _service.GetCollection<dynamic>(collection)
                                 .Find(_ => true)
                                 .ToListAsync();
        return Ok(data);
    }

    [HttpPost("{collection}")]
    public async Task<IActionResult> Post(string collection, [FromBody] dynamic document)
    {
        await _service.GetCollection<dynamic>(collection).InsertOneAsync(document);
        return Ok();
    }
}
