Prerequisites
Azure Key Vault set up with secrets.
Azure Managed Identity enabled for your service.
A database (SQL/MongoDB/etc.) that maps customerId to a KeyVaultName.
Azure SDK for .NET (Azure.Security.KeyVault.Secrets and Azure.Identity packages installed).

    public string KeyVaultName { get; set; }  // Key Vault name

Create a Service to Fetch Key Vault Secrets
This service:

Gets the Key Vault name from a database.
Uses Managed Identity to connect and retrieve secrets dynamically.

using Azure.Identity;
using Azure.Security.KeyVault.Secrets;
using Microsoft.Extensions.Caching.Memory;
using MongoDB.Driver; 

public class KeyVaultService
{
    private readonly IMongoCollection<CustomerConfig> _customerConfigs;
    private readonly IMemoryCache _cache; // Caching to reduce Key Vault API calls

    public KeyVaultService(IMongoClient mongoClient, IMemoryCache cache)
    {
        var database = mongoClient.GetDatabase("CustomerDatabase");
        _customerConfigs = database.GetCollection<CustomerConfig>("CustomerConfigs");
        _cache = cache;
    }

    public async Task<string> GetSecretAsync(string customerId, string secretName)
    {
        // Try cache first to reduce DB and Key Vault calls
        if (_cache.TryGetValue($"{customerId}-{secretName}", out string cachedSecret))
        {
            return cachedSecret;
        }

        // Fetch the Key Vault name for the customer
        var customer = await _customerConfigs.Find(t => t.CustomerId == customerId).FirstOrDefaultAsync();
        if (customer == null)
        {
            throw new Exception("customer configuration not found");
        }

        string keyVaultUrl = $"https://{customer.KeyVaultName}.vault.azure.net/";

        // Connect to Key Vault using Managed Identity
        var client = new SecretClient(new Uri(keyVaultUrl), new DefaultAzureCredential());
        var secret = await client.GetSecretAsync(secretName);

        // Store in cache for 10 minutes
        _cache.Set($"{customerId}-{secretName}", secret.Value.Value, TimeSpan.FromMinutes(10));

        return secret.Value.Value;
    }
}

builder.Services.AddScoped<KeyVaultService>();



Use the Service in a Controller
A simple API endpoint that retrieves secrets dynamically.


using Microsoft.AspNetCore.Mvc;

[ApiController]
[Route("api/[controller]")]
public class SecretController : ControllerBase
{
    private readonly KeyVaultService _keyVaultService;

    public SecretController(KeyVaultService keyVaultService)
    {
        _keyVaultService = keyVaultService;
    }

    [HttpGet("get-secret")]
    public async Task<IActionResult> GetSecret([FromHeader] string customerId, [FromQuery] string secretName)
    {
        if (string.IsNullOrEmpty(customerId) || string.IsNullOrEmpty(secretName))
        {
            return BadRequest("Missing parameters");
        }

        var secretValue = await _keyVaultService.GetSecretAsync(customerId, secretName);
        return Ok(new { SecretValue = secretValue });
    }
}

using Azure.Identity;
using Azure.Security.KeyVault.Secrets;
using Microsoft.Extensions.Caching.Memory;
using MongoDB.Driver;
using Microsoft.Extensions.Logging;

public class SecretService
{
    private readonly IMemoryCache _cache;
    private readonly SecretClient _keyVaultClient;
    private readonly ILogger<SecretService> _logger;

    public SecretService(IConfiguration configuration, IMemoryCache cache, ILogger<SecretService> logger)
    {
        _cache = cache;
        _logger = logger;

        string keyVaultUrl = configuration["KeyVault:Url"];
        _keyVaultClient = new SecretClient(new Uri(keyVaultUrl), new DefaultAzureCredential());

        CacheGlobalAdminDbConnection();
    }

    private void CacheGlobalAdminDbConnection()
    {
        try
        {
            var globalDbSecret = _keyVaultClient.GetSecret("GlobalAdminDB").Value.Value;

            var cacheEntryOptions = new MemoryCacheEntryOptions()
                .SetAbsoluteExpiration(TimeSpan.FromHours(1)) // **Expires in 1 hour**
                .RegisterPostEvictionCallback((key, value, reason, state) =>
                {
                    if (reason == EvictionReason.Expired)
                    {
                        CacheGlobalAdminDbConnection(); // **Auto-renew when expired**
                    }
                });

            _cache.Set("GlobalAdminDB", globalDbSecret, cacheEntryOptions);

            _logger.LogInformation("Cached Global Admin DB Connection String.");
        }
        catch (Exception ex)
        {
            _logger.LogError($"Failed to fetch Global Admin DB Connection: {ex.Message}");
            throw;
        }
    }

    public string GetGlobalAdminDbConnection()
    {
        return _cache.TryGetValue("GlobalAdminDB", out string connectionString) 
            ? connectionString 
            : throw new Exception("⚠ Global Admin DB connection is missing in cache!");
    }
}

using Azure.Identity;
using Azure.Security.KeyVault.Secrets;
using Microsoft.Extensions.Caching.Memory;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;

public class GlobalDbSecretRefreshService : BackgroundService
{
    private readonly IMemoryCache _cache;
    private readonly SecretClient _keyVaultClient;
    private readonly ILogger<GlobalDbSecretRefreshService> _logger;

    public GlobalDbSecretRefreshService(IConfiguration configuration, IMemoryCache cache, ILogger<GlobalDbSecretRefreshService> logger)
    {
        _cache = cache;
        _logger = logger;

        string keyVaultUrl = configuration["KeyVault:Url"];
        _keyVaultClient = new SecretClient(new Uri(keyVaultUrl), new DefaultAzureCredential());
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                await RefreshGlobalAdminDbConnection();
            }
            catch (Exception ex)
            {
                _logger.LogError($"Error refreshing Global Admin DB connection: {ex.Message}");
            }

            // Wait for 55 minutes before refreshing again
            await Task.Delay(TimeSpan.FromMinutes(55), stoppingToken);
        }
    }

    private async Task RefreshGlobalAdminDbConnection()
    {
        var secret = await _keyVaultClient.GetSecretAsync("GlobalAdminDB");
        string newConnectionString = secret.Value.Value;

        _cache.Set("GlobalAdminDB", newConnectionString, new MemoryCacheEntryOptions()
            .SetAbsoluteExpiration(TimeSpan.FromHours(1))); // Keep it fresh for 1 hour

        _logger.LogInformation("Refreshed Global Admin DB Connection String.");
    }
}

The global admin DB connection string is stored in Key Vault. When the app starts, it fetches the connection string, creates a MongoDB client, and caches it.

builder.Services.AddSingleton<IMemoryCache, MemoryCache>();
builder.Services.AddSingleton<IMongoClient>(serviceProvider =>
{
    var secretClient = serviceProvider.GetRequiredService<SecretClient>();
    var cache = serviceProvider.GetRequiredService<IMemoryCache>();

    // Fetch MongoDB connection string from Key Vault
    KeyVaultSecret secret = secretClient.GetSecret("GlobalAdminDBConnectionString");
    string connectionString = secret.Value;

    if (string.IsNullOrEmpty(connectionString))
    {
        throw new InvalidOperationException("Failed to retrieve the MongoDB connection string from Key Vault.");
    }

    // Store in memory cache with 55-minute expiration
    cache.Set("GlobalAdminDB", connectionString, new MemoryCacheEntryOptions
    {
        AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(55)
    });

    return new MongoClient(connectionString);
});

using System;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Caching.Memory;
using Azure.Security.KeyVault.Secrets;
using MongoDB.Driver;

public class MongoDbCacheRefreshService : BackgroundService
{
    private readonly IMemoryCache _cache;
    private readonly SecretClient _secretClient;
    private readonly IServiceProvider _serviceProvider;

    public MongoDbCacheRefreshService(IMemoryCache cache, SecretClient secretClient, IServiceProvider serviceProvider)
    {
        _cache = cache;
        _secretClient = secretClient;
        _serviceProvider = serviceProvider;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                KeyVaultSecret secret = await _secretClient.GetSecretAsync("GlobalAdminDBConnectionString");
                string connectionString = secret.Value;

                if (!string.IsNullOrEmpty(connectionString))
                {
                    _cache.Set("GlobalAdminDB", connectionString, new MemoryCacheEntryOptions
                    {
                        AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(55)
                    });

                    // Optionally, recreate and replace the MongoClient
                    using var scope = _serviceProvider.CreateScope();
                    var mongoClient = new MongoClient(connectionString);
                    _cache.Set("MongoClient", mongoClient);
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Error refreshing MongoDB connection: {ex.Message}");
            }

            await Task.Delay(TimeSpan.FromMinutes(55), stoppingToken);
        }
    }
}



builder.Services.AddHostedService<MongoDbCacheRefreshService>();



Retrieving Cached MongoDB Client in Controllers
Anywhere in your WebAPI controllers or services, retrieve the MongoDB client dynamically like this:

public class MyService
{
    private readonly IMemoryCache _cache;

    public MyService(IMemoryCache cache)
    {
        _cache = cache;
    }

    public IMongoDatabase GetDatabase()
    {
        if (_cache.TryGetValue("GlobalAdminDB", out string connectionString))
        {
            var client = new MongoClient(connectionString);
            return client.GetDatabase("YourDatabaseName");
        }

        throw new InvalidOperationException("MongoDB connection is not available.");
    }
}

using System.Collections.Generic;
using System.Threading.Tasks;
using MongoDB.Driver;
using Microsoft.Extensions.Caching.Memory;

public class CollectionConfigService
{
    private readonly IMongoDatabase _globalDb;
    private readonly IMemoryCache _cache;
    
    public CollectionConfigService(IMongoClient globalClient, IMemoryCache cache)
    {
        _globalDb = globalClient.GetDatabase("GlobalAdminDB"); // Change to actual global DB name
        _cache = cache;
    }

    public async Task LoadCollectionConfigAsync()
    {
        var configCollection = _globalDb.GetCollection<TenantConfig>("TenantConfig");
        var config = await configCollection.Find(c => c.Id == "global-config").FirstOrDefaultAsync();

        if (config != null)
        {
            _cache.Set("SystemCollections", config.SystemCollections);
            _cache.Set("TenantCollections", config.TenantCollections);
            _cache.Set("TenantMappings", config.Tenants);
        }
    }

    public bool IsGlobalCollection(string collectionName)
    {
        return _cache.TryGetValue("SystemCollections", out List<string> systemCollections) && 
               systemCollections.Contains(collectionName);
    }

    public bool IsTenantCollection(string collectionName)
    {
        return _cache.TryGetValue("TenantCollections", out List<string> tenantCollections) &&
               tenantCollections.Contains(collectionName);
    }

    public string GetTenantDatabase(string tenantId)
    {
        if (_cache.TryGetValue("TenantMappings", out List<TenantConfig.TenantInfo> tenants))
        {
            var tenant = tenants.Find(t => t.TenantId == tenantId);
            return tenant?.DatabaseName;
        }
        return null;
    }
}

///[ApiController]
[Route("api/[controller]")]
public class DataController : ControllerBase
{
    private readonly CollectionConfigService _configService;
    private readonly IMongoClient _globalClient;
    private readonly IMemoryCache _cache;
    
    public DataController(CollectionConfigService configService, IMongoClient globalClient, IMemoryCache cache)
    {
        _configService = configService;
        _globalClient = globalClient;
        _cache = cache;
    }

    [HttpPost("fetch")]
    public IActionResult FetchData([FromHeader] string customerId, [FromBody] FetchRequest request)
    {
        string collectionName = request.CollectionName;

        if (_configService.IsGlobalCollection(collectionName))
        {
            var globalDb = _globalClient.GetDatabase("GlobalAdminDB");
            var collection = globalDb.GetCollection<BsonDocument>(collectionName);
            var result = collection.Find(FilterDefinition<BsonDocument>.Empty).ToList();
            return Ok(result);
        }
        else if (_configService.IsTenantCollection(collectionName))
        {
            string tenantDbName = _configService.GetTenantDatabase(customerId);
            if (string.IsNullOrEmpty(tenantDbName))
                return BadRequest("Invalid tenant ID");

            if (!_cache.TryGetValue($"TenantDB-{customerId}", out IMongoClient tenantClient))
            {
                // Get connection string from Key Vault and create new MongoClient
                var secretClient = new SecretClient(new Uri("https://your-keyvault.vault.azure.net/"), new DefaultAzureCredential());
                var secret = secretClient.GetSecret($"Tenant-{customerId}-DB-Connection");
                tenantClient = new MongoClient(secret.Value);

                // Cache the client
                _cache.Set($"TenantDB-{customerId}", tenantClient, new MemoryCacheEntryOptions
                {
                    AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(55)
                });
            }

            var tenantDb = tenantClient.GetDatabase(tenantDbName);
            var collection = tenantDb.GetCollection<BsonDocument>(collectionName);
            var result = collection.Find(FilterDefinition<BsonDocument>.Empty).ToList();

            return Ok(result);
        }

        return BadRequest("Unknown collection reference");
    }
}
