[ApiController]
[Route("api/connection")]
public class ConnectionController : ControllerBase
{
    private readonly IMemoryCache _cache;
    private readonly MongoClientService _mongoService;

    public ConnectionController(IMemoryCache cache, MongoClientService mongoService)
    {
        _cache = cache;
        _mongoService = mongoService;
    }

    [HttpPost("refresh")]
    public async Task<IActionResult> RefreshMongoConnection()
    {
        // Remove cached MongoDB connection string
        _cache.Remove("GlobalDbConnection");

        // Reinitialize MongoDB connection
        await _mongoService.ReinitializeMongoClient();

        return Ok(new { message = "MongoDB connection refreshed successfully." });
    }
}



public class MongoClientService
{
    private readonly IMemoryCache _cache;
    private readonly SecretClient _secretClient;
    private MongoClient _mongoClient;

    public MongoClientService(IMemoryCache cache, SecretClient secretClient)
    {
        _cache = cache;
        _secretClient = secretClient;
    }

    public async Task ReinitializeMongoClient()
    {
        var mongoUri = await _secretClient.GetSecretAsync("GlobalMongoConnectionString");

        _mongoClient = new MongoClient(mongoUri.Value.Value);
        _cache.Set("GlobalDbConnection", _mongoClient, TimeSpan.FromMinutes(55));
    }

    public IMongoDatabase GetGlobalDatabase()
    {
        if (!_cache.TryGetValue("GlobalDbConnection", out MongoClient client))
        {
            throw new Exception("MongoDB connection is not initialized.");
        }
        return client.GetDatabase("GlobalAdminDB");
    }
}

using Azure.Security.KeyVault.Secrets;
using Azure.Identity;
using Microsoft.Extensions.Caching.Memory;
using Microsoft.Extensions.Configuration;
using MongoDB.Driver;
using System;
using System.Threading.Tasks;

public class MongoDbService
{
    private readonly IConfiguration _configuration;
    private readonly SecretClient _keyVaultClient;
    private readonly IMemoryCache _cache;
    private IMongoClient _mongoClient;
    private IMongoDatabase _database;
    private readonly string _cacheKey = "MongoDbConnection";

    public MongoDbService(IConfiguration configuration, IMemoryCache cache)
    {
        _configuration = configuration;
        _cache = cache;

        string environment = _configuration["ASPNETCORE_ENVIRONMENT"];

        if (environment != "local")
        {
            string keyVaultUrl = _configuration["KeyVault:URL"];
            _keyVaultClient = new SecretClient(new Uri(keyVaultUrl), new DefaultAzureCredential());
        }

        // Initialize MongoDB connection
        LoadMongoConnection();
    }

    private void LoadMongoConnection()
    {
        string connectionString = GetMongoConnectionString().Result;
        _cache.Set(_cacheKey, connectionString, TimeSpan.FromMinutes(55));

        _mongoClient = new MongoClient(connectionString);
        _database = _mongoClient.GetDatabase(_configuration["MongoSettings:DatabaseName"]);
    }

    private async Task<string> GetMongoConnectionString()
    {
        string environment = _configuration["ASPNETCORE_ENVIRONMENT"];

        if (environment == "local")
        {
            return _configuration["MongoSettings:ConnectionString"];
        }
        else
        {
            KeyVaultSecret secret = await _keyVaultClient.GetSecretAsync("GlobalMongoConnectionString");
            return secret.Value;
        }
    }

    public IMongoDatabase GetDatabase() => _database;

    public void RefreshMongoConnection()
    {
        LoadMongoConnection();
    }
}

public class MongoDbService : IMongoDbService
{
    private MongoClient _mongoClient;
    private readonly IKeyVaultService _keyVaultService;
    private readonly IConfiguration _configuration;
    private readonly object _lock = new();

    public MongoDbService(IKeyVaultService keyVaultService, IConfiguration configuration)
    {
        _keyVaultService = keyVaultService;
        _configuration = configuration;
        LoadMongoConnection();
    }

    private void LoadMongoConnection()
    {
        try
        {
            string connectionString = GetMongoConnectionString().Result;

            lock (_lock)
            {
                if (_mongoClient != null)
                {
                    _mongoClient.Cluster.Dispose(); // Dispose old connection
                }

                _mongoClient = new MongoClient(connectionString);
            }
        }
        catch (Exception ex)
        {
            throw new Exception("An error occurred while establishing the MongoDB connection.", ex);
        }
    }

    private async Task<string> GetMongoConnectionString()
    {
        // Use appsettings for local development
        if (_configuration["Environment"] == "Local")
        {
            return _configuration.GetConnectionString("MongoDB");
        }

        // Fetch from Key Vault in non-local environments
        return await _keyVaultService.GetSecretAsync("MongoDBConnectionString");
    }

    public void RefreshMongoConnection()
    {
        LoadMongoConnection();
    }

    public IMongoDatabase GetDatabase(string databaseName)
    {
        return _mongoClient.GetDatabase(databaseName);
    }
}



private readonly object _lock = new(); // Lock object for thread safety

private void LoadMongoConnection()
{
    try
    {
        string connectionString = GetMongoConnectionString().Result;

        lock (_lock)  // Ensure only one thread updates the MongoClient at a time
        {
            if (_mongoClient != null)
            {
                _mongoClient.Cluster.Dispose(); // Dispose old client safely
            }

            _mongoClient = new MongoClient(connectionString); // Set new client
        }
    }
    catch (Exception ex)
    {
        throw new Exception("An error occurred while establishing the MongoDB connection.", ex);
    }
}
