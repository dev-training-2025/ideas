using Microsoft.AspNetCore.Mvc;
using MongoDB.Bson;
using MongoDB.Driver;
using MongoAPI.Models;
using MongoAPI.Services;

namespace MongoAPI.Controllers
{
    [ApiController]
    [Route("action")]
    public class MongoDbController : ControllerBase
    {
        private readonly IMongoDbService _mongoService;

        public MongoDbController(IMongoDbService mongoService)
        {
            _mongoService = mongoService;
        }

        [HttpPost("findOne")]
        public async Task<ActionResult<FindOneResponse>> FindOne([FromBody] FindOneRequest request)
        {
            var filter = BsonDocument.Parse(request.Filter.ToString());
            var projection = request.Projection != null 
                ? BsonDocument.Parse(request.Projection.ToString()) 
                : null;

            var result = await _mongoService.FindOneAsync(
                request.Database,
                request.Collection,
                filter,
                projection
            );

            return Ok(new FindOneResponse { Document = result });
        }

        [HttpPost("find")]
        public async Task<ActionResult<FindResponse>> Find([FromBody] FindRequest request)
        {
            var filter = BsonDocument.Parse(request.Filter.ToString());
            var projection = request.Projection != null 
                ? BsonDocument.Parse(request.Projection.ToString()) 
                : null;
            var sort = request.Sort != null 
                ? BsonDocument.Parse(request.Sort.ToString()) 
                : null;

            var results = await _mongoService.FindManyAsync(
                request.Database,
                request.Collection,
                filter,
                projection,
                sort,
                request.Skip,
                request.Limit
            );

            return Ok(new FindResponse { Documents = results });
        }

        [HttpPost("insertOne")]
        public async Task<ActionResult<InsertOneResponse>> InsertOne([FromBody] InsertOneRequest request)
        {
            var document = BsonDocument.Parse(request.Document.ToString());
            var id = await _mongoService.InsertOneAsync(
                request.Database,
                request.Collection,
                document
            );

            return Ok(new InsertOneResponse { InsertedId = id });
        }

        [HttpPost("insertMany")]
        public async Task<ActionResult<InsertManyResponse>> InsertMany([FromBody] InsertManyRequest request)
        {
            var documents = request.Documents.Select(doc => BsonDocument.Parse(doc.ToString()));
            var ids = await _mongoService.InsertManyAsync(
                request.Database,
                request.Collection,
                documents
            );

            return Ok(new InsertManyResponse { InsertedIds = ids });
        }

        [HttpPost("updateOne")]
        public async Task<ActionResult<UpdateResponse>> UpdateOne([FromBody] UpdateRequest request)
        {
            var filter = BsonDocument.Parse(request.Filter.ToString());
            var update = BsonDocument.Parse(request.Update.ToString());

            var result = await _mongoService.UpdateOneAsync(
                request.Database,
                request.Collection,
                filter,
                update,
                request.Upsert
            );

            return Ok(new UpdateResponse 
            { 
                MatchedCount = result.MatchedCount,
                ModifiedCount = result.ModifiedCount,
                UpsertedId = result.UpsertedId?.ToString()
            });
        }

        [HttpPost("updateMany")]
        public async Task<ActionResult<UpdateResponse>> UpdateMany([FromBody] UpdateRequest request)
        {
            var filter = BsonDocument.Parse(request.Filter.ToString());
            var update = BsonDocument.Parse(request.Update.ToString());

            var result = await _mongoService.UpdateManyAsync(
                request.Database,
                request.Collection,
                filter,
                update,
                request.Upsert
            );

            return Ok(new UpdateResponse 
            { 
                MatchedCount = result.MatchedCount,
                ModifiedCount = result.ModifiedCount,
                UpsertedId = result.UpsertedId?.ToString()
            });
        }

        [HttpPost("deleteOne")]
        public async Task<ActionResult<DeleteResponse>> DeleteOne([FromBody] DeleteRequest request)
        {
            var filter = BsonDocument.Parse(request.Filter.ToString());
            var count = await _mongoService.DeleteOneAsync(
                request.Database,
                request.Collection,
                filter
            );

            return Ok(new DeleteResponse { DeletedCount = count });
        }

        [HttpPost("deleteMany")]
        public async Task<ActionResult<DeleteResponse>> DeleteMany([FromBody] DeleteRequest request)
        {
            var filter = BsonDocument.Parse(request.Filter.ToString());
            var count = await _mongoService.DeleteManyAsync(
                request.Database,
                request.Collection,
                filter
            );

            return Ok(new DeleteResponse { DeletedCount = count });
        }

        [HttpPost("aggregate")]
        public async Task<ActionResult<AggregateResponse>> Aggregate([FromBody] AggregateRequest request)
        {
            var pipeline = request.Pipeline.Select(stage => BsonDocument.Parse(stage.ToString()));
            var results = await _mongoService.AggregateAsync(
                request.Database,
                request.Collection,
                pipeline
            );

            return Ok(new AggregateResponse { Documents = results });
        }
    }
}





MongoDB Service Implementation

using MongoDB.Bson;
using MongoDB.Driver;

namespace MongoAPI.Services
{
    public interface IMongoDbService
    {
        Task<object> FindOneAsync(string database, string collection, BsonDocument filter, BsonDocument projection = null);
        Task<object[]> FindManyAsync(string database, string collection, BsonDocument filter, BsonDocument projection = null, BsonDocument sort = null, int? skip = null, int? limit = null);
        Task<string> InsertOneAsync(string database, string collection, BsonDocument document);
        Task<string[]> InsertManyAsync(string database, string collection, IEnumerable<BsonDocument> documents);
        Task<UpdateResult> UpdateOneAsync(string database, string collection, BsonDocument filter, BsonDocument update, bool upsert = false);
        Task<UpdateResult> UpdateManyAsync(string database, string collection, BsonDocument filter, BsonDocument update, bool upsert = false);
        Task<long> DeleteOneAsync(string database, string collection, BsonDocument filter);
        Task<long> DeleteManyAsync(string database, string collection, BsonDocument filter);
        Task<object[]> AggregateAsync(string database, string collection, IEnumerable<BsonDocument> pipeline);
    }

    public class MongoDbService : IMongoDbService
    {
        private readonly IMongoClient _client;

        public MongoDbService(IConfiguration configuration)
        {
            var connectionString = configuration.GetValue<string>("MongoDB:ConnectionString");
            _client = new MongoClient(connectionString);
        }

        private IMongoCollection<BsonDocument> GetCollection(string database, string collection)
        {
            return _client
                .GetDatabase(database)
                .GetCollection<BsonDocument>(collection);
        }

        public async Task<object> FindOneAsync(string database, string collection, BsonDocument filter, BsonDocument projection = null)
        {
            var options = new FindOptions<BsonDocument>
            {
                Projection = projection
            };

            var result = await GetCollection(database, collection)
                .Find(filter)
                .FirstOrDefaultAsync();

            return result?.ToJson();
        }

        public async Task<object[]> FindManyAsync(string database, string collection, BsonDocument filter, BsonDocument projection = null, BsonDocument sort = null, int? skip = null, int? limit = null)
        {
            var options = new FindOptions<BsonDocument>
            {
                Projection = projection,
                Sort = sort,
                Skip = skip,
                Limit = limit
            };

            var results = await GetCollection(database, collection)
                .Find(filter)
                .ToListAsync();

            return results.Select(doc => doc.ToJson()).ToArray();
        }

        public async Task<string> InsertOneAsync(string database, string collection, BsonDocument document)
        {
            await GetCollection(database, collection)
                .InsertOneAsync(document);

            return document["_id"].ToString();
        }

        public async Task<string[]> InsertManyAsync(string database, string collection, IEnumerable<BsonDocument> documents)
        {
            await GetCollection(database, collection)
                .InsertManyAsync(documents);

            return documents.Select(doc => doc["_id"].ToString()).ToArray();
        }

        public async Task<UpdateResult> UpdateOneAsync(string database, string collection, BsonDocument filter, BsonDocument update, bool upsert = false)
        {
            var options = new UpdateOptions { IsUpsert = upsert };
            return await GetCollection(database, collection)
                .UpdateOneAsync(filter, update, options);
        }

        public async Task<UpdateResult> UpdateManyAsync(string database, string collection, BsonDocument filter, BsonDocument update, bool upsert = false)
        {
            var options = new UpdateOptions { IsUpsert = upsert };
            return await GetCollection(database, collection)
                .UpdateManyAsync(filter, update, options);
        }

        public async Task<long> DeleteOneAsync(string database, string collection, BsonDocument filter)
        {
            var result = await GetCollection(database, collection)
                .DeleteOneAsync(filter);
            return result.DeletedCount;
        }

        public async Task<long> DeleteManyAsync(string database, string collection, BsonDocument filter)
        {
            var result = await GetCollection(database, collection)
                .DeleteManyAsync(filter);
            return result.DeletedCount;
        }

        public async Task<object[]> AggregateAsync(string database, string collection, IEnumerable<BsonDocument> pipeline)
        {
            var results = await GetCollection(database, collection)
                .Aggregate<BsonDocument>(pipeline.ToList())
                .ToListAsync();

            return results.Select(doc => doc.ToJson()).ToArray();
        }
    }
}




//middleware

public class RequestValidationMiddleware
{
    private readonly RequestDelegate _next;

    public RequestValidationMiddleware(RequestDelegate next)
    {
        _next = next;
    }

    public async Task Invoke(HttpContext context)
    {
        // Check if the request content type is JSON
        if (!context.Request.ContentType?.Contains("application/json") == true)
        {
            context.Response.StatusCode = StatusCodes.Status400BadRequest;
            await context.Response.WriteAsync("Invalid content type. Expected application/json.");
            return;
        }

        // Check if CustomerId header exists
        if (!context.Request.Headers.TryGetValue("CustomerId", out var customerId) || string.IsNullOrWhiteSpace(customerId))
        {
            context.Response.StatusCode = StatusCodes.Status400BadRequest;
            await context.Response.WriteAsync("Missing or invalid CustomerId header.");
            return;
        }

        // Store the Customer ID in HttpContext for later retrieval
        context.Items["CustomerId"] = customerId.ToString();

        await _next(context);
    }
}



//Base validation
public abstract class BaseRequestValidator<T> : AbstractValidator<T> where T : CommonRequest
{
    public BaseRequestValidator()
    {
        RuleFor(x => x.DatabaseName)
            .NotEmpty().WithMessage("DatabaseName is required.")
            .MaximumLength(100);

        RuleFor(x => x.CollectionName)
            .NotEmpty().WithMessage("CollectionName is required.")
            .MaximumLength(100);
    }
}

//Inherit in Other Validators
public class InsertOneRequestValidator : BaseRequestValidator<InsertOneRequest>
{
    public InsertOneRequestValidator()
    {
        RuleFor(x => x.Document)
            .NotNull().WithMessage("Document is required.");
    }
}

public class UpdateOneRequestValidator : BaseRequestValidator<UpdateOneRequest>
{
    public UpdateOneRequestValidator()
    {
        RuleFor(x => x.UpdateFields)
            .NotNull().WithMessage("UpdateFields is required.");
    }
}

2. Retrieve Connection String Securely from Azure Key Vault

public interface ICustomerConfigService
{
    Task<string?> GetDatabaseNameAsync(string customerId);
    Task<string?> GetBlobStorageConnectionStringAsync(string customerId);
}

public class CustomerConfigService : ICustomerConfigService
{
    private readonly IMongoCollection<CustomerConfig> _customerConfigCollection;
    private readonly SecretClient _secretClient;

    public CustomerConfigService(IMongoClient mongoClient, SecretClient secretClient)
    {
        var database = mongoClient.GetDatabase("ConfigDB"); // Static config DB
        _customerConfigCollection = database.GetCollection<CustomerConfig>("CustomerConfigs");
        _secretClient = secretClient;
    }

    public async Task<string?> GetDatabaseNameAsync(string customerId)
    {
        var filter = Builders<CustomerConfig>.Filter.Eq(c => c.CustomerId, customerId);
        var customerConfig = await _customerConfigCollection.Find(filter).FirstOrDefaultAsync();
        return customerConfig?.DatabaseName;
    }

    public async Task<string?> GetBlobStorageConnectionStringAsync(string customerId)
    {
        var filter = Builders<CustomerConfig>.Filter.Eq(c => c.CustomerId, customerId);
        var customerConfig = await _customerConfigCollection.Find(filter).FirstOrDefaultAsync();
        if (customerConfig == null || string.IsNullOrEmpty(customerConfig.KeyVaultSecretName))
        {
            return null; // No storage config found
        }

        // Retrieve secret from Azure Key Vault
        KeyVaultSecret secret = await _secretClient.GetSecretAsync(customerConfig.KeyVaultSecretName);
        return secret.Value;
    }
}

4. Securely Access Blob Storage in Controller
In your file upload controller, retrieve the connection string from Key Vault before uploading files.

[HttpPost("upload")]
public async Task<IActionResult> UploadFile(
    [FromHeader(Name = "customerId")] string customerId,
    IFormFile file)
{
    if (string.IsNullOrEmpty(customerId))
    {
        return BadRequest("Missing customerId in headers.");
    }

    var connectionString = await _customerConfigService.GetBlobStorageConnectionStringAsync(customerId);
    if (string.IsNullOrEmpty(connectionString))
    {
        return NotFound("Blob Storage connection not found for customer.");
    }

    // Upload to Azure Blob Storage
    var blobServiceClient = new BlobServiceClient(connectionString);
    var containerClient = blobServiceClient.GetBlobContainerClient("customer-files");
    var blobClient = containerClient.GetBlobClient(file.FileName);

    using (var stream = file.OpenReadStream())
    {
        await blobClient.UploadAsync(stream, true);
    }

    return Ok("File uploaded successfully.");
}
