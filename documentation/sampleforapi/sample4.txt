Interface: IGlobalDbService

public interface IGlobalDbService
{
    /// <summary>
    /// Returns the global "admin" database for reading tenant config, etc.
    /// </summary>
    IMongoDatabase GetGlobalDatabase();

    /// <summary>
    /// Refreshes the underlying MongoDB client if the connection string changes.
    /// </summary>
    void RefreshGlobalConnection();
}

Implementation: GlobalDbService

using MongoDB.Driver;
using Microsoft.Extensions.Configuration;
using System;

public class GlobalDbService : IGlobalDbService
{
    private MongoClient _mongoClient;
    private readonly IConfiguration _configuration;
    private readonly object _lock = new();

    public GlobalDbService(IConfiguration configuration)
    {
        _configuration = configuration;
        LoadGlobalConnection();
    }

    private void LoadGlobalConnection()
    {
        lock (_lock)
        {
            // 1. Fetch connection string from local or Key Vault
            string connectionString = GetGlobalConnectionString();

            // 2. Create a new MongoClient
            _mongoClient = new MongoClient(connectionString);
        }
    }

    private string GetGlobalConnectionString()
    {
        // If local environment
        if (_configuration["Environment"] == "Local")
        {
            return _configuration.GetConnectionString("GlobalAdminMongo");
        }
        else
        {
            // Production example: fetch from Key Vault or other secure source
            // return keyVaultService.GetSecret("GlobalAdminMongoConnectionString");
            return "mongodb://someProductionUri";
        }
    }

    public IMongoDatabase GetGlobalDatabase()
    {
        // Example DB name "GlobalAdminDB"
        return _mongoClient.GetDatabase("GlobalAdminDB");
    }

    public void RefreshGlobalConnection()
    {
        LoadGlobalConnection();
    }
}
Key Points

No GetGlobalClient() method; the client is internal.
Controllers/services just call GetGlobalDatabase().
RefreshGlobalConnection() if you need to forcibly recreate the client.



Interface: ITenantDbService

public interface ITenantDbService
{
    /// <summary>
    /// Returns a tenant-specific database by name.
    /// </summary>
    IMongoDatabase GetTenantDatabase(string tenantId, string dbName);

    /// <summary>
    /// Refresh the tenant’s MongoDB client if the tenant’s connection changes.
    /// </summary>
    void RefreshTenantConnection(string tenantId);
}
Implementation: TenantDbService

using MongoDB.Driver;
using System.Collections.Generic;
using System;

public class TenantDbService : ITenantDbService
{
    private readonly IGlobalDbService _globalDbService;
    private readonly Dictionary<string, MongoClient> _tenantClients = new();
    private readonly object _lock = new();

    public TenantDbService(IGlobalDbService globalDbService)
    {
        _globalDbService = globalDbService;
    }

    public IMongoDatabase GetTenantDatabase(string tenantId, string dbName)
    {
        var client = GetOrCreateTenantClient(tenantId);
        return client.GetDatabase(dbName);
    }

    public void RefreshTenantConnection(string tenantId)
    {
        lock (_lock)
        {
            // Remove the old client so next request re-fetches
            _tenantClients.Remove(tenantId);
        }
    }

    private MongoClient GetOrCreateTenantClient(string tenantId)
    {
        lock (_lock)
        {
            if (_tenantClients.TryGetValue(tenantId, out var existingClient))
            {
                return existingClient;
            }

            // 1. Fetch tenant config from the global admin DB
            var tenantConfig = FetchTenantConfig(tenantId);

            // 2. Create new client for this tenant
            var client = new MongoClient(tenantConfig.ConnectionString);

            _tenantClients[tenantId] = client;
            return client;
        }
    }

    private TenantConfig FetchTenantConfig(string tenantId)
    {
        var globalDb = _globalDbService.GetGlobalDatabase();
        var tenantsColl = globalDb.GetCollection<TenantConfig>("Tenants");
        return tenantsColl.Find(t => t.TenantId == tenantId).FirstOrDefault();
    }
}

// Example model for tenant config
public class TenantConfig
{
    public string TenantId { get; set; }
    public string ConnectionString { get; set; }
}
Key Points

No GetTenantClient() method is exposed; the service internally manages a dictionary of MongoClients.
External code just calls GetTenantDatabase(tenantId, dbName).
RefreshTenantConnection(tenantId) removes the old client so that next time you request the database, it fetches a new connection.



3. Usage in a Controller

[ApiController]
[Route("api/tenant-data")]
public class TenantDataController : ControllerBase
{
    private readonly ITenantDbService _tenantDbService;

    public TenantDataController(ITenantDbService tenantDbService)
    {
        _tenantDbService = tenantDbService;
    }

    [HttpGet("{tenantId}/{dbName}/orders")]
    public IActionResult GetTenantOrders(string tenantId, string dbName)
    {
        var db = _tenantDbService.GetTenantDatabase(tenantId, dbName);
        var ordersColl = db.GetCollection<BsonDocument>("Orders");
        var orders = ordersColl.Find(FilterDefinition<BsonDocument>.Empty).ToList();
        return Ok(orders);
    }
}

No direct references to MongoClient—just a call to GetTenantDatabase(tenantId, dbName).

builder.Services.AddSingleton<IGlobalDbService, GlobalDbService>();
builder.Services.AddSingleton<ITenantDbService, TenantDbService>();

using MongoDB.Driver;
using System;
using System.Collections.Concurrent;
using System.Threading.Tasks;

namespace DataStore.Services
{
    public class TenantDbService : ITenantDbService
    {
        private readonly IGlobalDbService _globalDbService;
        private readonly IKeyVaultService _keyVaultService;
        private readonly ConcurrentDictionary<string, IMongoClient> _tenantClients;

        public TenantDbService(IGlobalDbService globalDbService, IKeyVaultService keyVaultService)
        {
            _globalDbService = globalDbService;
            _keyVaultService = keyVaultService;
            _tenantClients = new ConcurrentDictionary<string, IMongoClient>();
        }

        public async Task<IMongoDatabase> GetTenantDatabaseAsync(string tenantId)
        {
            // Step 1: Fetch the key vault secret name from the global database
            var tenantConfigCollection = _globalDbService.GetDatabase().GetCollection<TenantConfig>("Tenants");
            var tenantConfig = await tenantConfigCollection.Find(t => t.TenantId == tenantId).FirstOrDefaultAsync();

            if (tenantConfig == null || string.IsNullOrEmpty(tenantConfig.MongoSecretName))
            {
                throw new Exception($"Tenant configuration not found for {tenantId}");
            }

            // Step 2: Retrieve the actual connection string from Key Vault
            string connectionString = await _keyVaultService.GetSecretAsync(tenantConfig.MongoSecretName);

            if (string.IsNullOrEmpty(connectionString))
            {
                throw new Exception($"MongoDB connection string is missing for tenant {tenantId}");
            }

            // Step 3: Reuse or create a new MongoClient instance
            var mongoClient = _tenantClients.GetOrAdd(tenantId, _ => new MongoClient(connectionString));

            // Step 4: Return the tenant database
            return mongoClient.GetDatabase(tenantConfig.DatabaseName);
        }
    }

    public interface ITenantDbService
    {
        Task<IMongoDatabase> GetTenantDatabaseAsync(string tenantId);
    }

    public class TenantConfig
    {
        public string TenantId { get; set; }
        public string MongoSecretName { get; set; }
        public string DatabaseName { get; set; }
    }
}
