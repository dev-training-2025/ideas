using System.Threading;
using System.Threading.Tasks;
using Xunit;
using Moq;
using MongoDB.Driver;
using System.Collections.Generic;

public class StructuredDataRepositoryTests
{
    private readonly Mock<IMongoCollection<TestDocument>> _mockCollection;
    private readonly Mock<IMongoDatabase> _mockDatabase;
    private readonly Mock<IMongoClient> _mockClient;
    private readonly StructuredDataRepository _repository;

    public StructuredDataRepositoryTests()
    {
        _mockCollection = new Mock<IMongoCollection<TestDocument>>();
        _mockDatabase = new Mock<IMongoDatabase>();
        _mockClient = new Mock<IMongoClient>();

        _mockDatabase.Setup(db => db.GetCollection<TestDocument>("TestCollection", null))
                     .Returns(_mockCollection.Object);

        _mockClient.Setup(client => client.GetDatabase(It.IsAny<string>(), null))
                   .Returns(_mockDatabase.Object);

        _repository = new StructuredDataRepository(_mockClient.Object);
    }

    [Fact]
    public async Task FindOneAsync_ShouldReturnDocument_WhenExcludeIsNotYes()
    {
        // Arrange
        var expectedDocument = new TestDocument { Id = "123", Name = "Valid Record", Exclude = "No" };
        var filter = Builders<TestDocument>.Filter.Eq(doc => doc.Id, "123");

        var mockCursor = new Mock<IAsyncCursor<TestDocument>>();
        mockCursor.SetupSequence(cursor => cursor.MoveNext(It.IsAny<CancellationToken>()))
                  .Returns(true)  // Data exists
                  .Returns(false); // No more data

        mockCursor.Setup(cursor => cursor.Current)
                  .Returns(new List<TestDocument> { expectedDocument });

        _mockCollection.Setup(col => col.FindAsync(
                                It.IsAny<FilterDefinition<TestDocument>>(),
                                It.IsAny<FindOptions<TestDocument, TestDocument>>(),
                                It.IsAny<CancellationToken>()))
                       .ReturnsAsync(mockCursor.Object);

        // Act
        var result = await _repository.FindOneAsync("TestCollection", filter);

        // Assert
        Assert.NotNull(result);
        Assert.Equal(expectedDocument.Id, result.Id);
    }

    [Fact]
    public async Task FindOneAsync_ShouldReturnNull_WhenExcludeIsYes()
    {
        // Arrange
        var deletedDocument = new TestDocument { Id = "123", Name = "Deleted Record", Exclude = "Yes" };
        var filter = Builders<TestDocument>.Filter.Eq(doc => doc.Id, "123");

        var mockCursor = new Mock<IAsyncCursor<TestDocument>>();
        mockCursor.SetupSequence(cursor => cursor.MoveNext(It.IsAny<CancellationToken>()))
                  .Returns(true)
                  .Returns(false);

        mockCursor.Setup(cursor => cursor.Current)
                  .Returns(new List<TestDocument> { deletedDocument });

        _mockCollection.Setup(col => col.FindAsync(
                                It.IsAny<FilterDefinition<TestDocument>>(),
                                It.IsAny<FindOptions<TestDocument, TestDocument>>(),
                                It.IsAny<CancellationToken>()))
                       .ReturnsAsync(mockCursor.Object);

        // Act
        var result = await _repository.FindOneAsync("TestCollection", filter);

        // Assert
        Assert.Null(result);
    }
}

Advantages of This Approach
Separation of Concerns:
Each collection has a single responsibility (e.g., user info, cust info, project info, role definitions, role assignments), which makes it easier to update and maintain.

Simplified Queries:
You can use MongoDB’s aggregation framework (with $lookup) to join these collections as needed. For example, to get the list of projects and roles a user has, you could aggregate the UserProjectRoles collection and then join with Users, Projects, and Roles.

Flexibility in Updates:
Changing a role’s definition, adding or removing a role from a user, or updating cust details doesn’t require modifying nested arrays across multiple documents.

Scalability:
This normalized approach tends to scale better for large datasets where users have access to many cust and projects, and where role assignments change frequently.

Easy to Enforce Access Control:
You can build queries like “find all UserProjectRoles where projectId = 'project_123' and roleIds contains 'role_a'” to quickly retrieve all annotaters for a project.


[
  { "$match": { "projectId": "project_123", "roleIds": "role_a" } },
  {
    "$lookup": {
      "from": "Users",
      "localField": "userId",
      "foreignField": "_id",
      "as": "userInfo"
    }
  },
  { "$unwind": "$userInfo" },
  {
    "$project": {
      "_id": 0,
      "userId": 1,
      "userName": "$userInfo.name",
      "email": "$userInfo.email"
    }
  }
]

This pipeline:

Filters the UserProjectRoles collection for entries for Project 123 with the role role_a.
Joins with the Users collection to get detailed user information.
Projects a concise result set.



Direct Aggregate Endpoint
Description:
Expose an API endpoint that accepts an aggregation pipeline (in JSON format) as part of the request. The endpoint then passes this pipeline to MongoDB’s aggregation framework and returns the results.

Advantages:

Flexibility: Clients can build complex queries (joins via $lookup, filtering with $match, grouping with $group, etc.) dynamically.
Unified API: One endpoint can serve multiple query needs instead of having several dedicated endpoints.
Considerations:

Security: Allowing arbitrary pipelines can be risky. You need to validate or sanitize the incoming pipeline to prevent misuse (e.g., disallow certain operators like $where or $function that might lead to performance or security issues).
Performance: Aggregation pipelines can be complex and resource-intensive. Implement limits on pipeline length, stage complexity, and runtime.
Validation: You may want to provide schema validation (perhaps using FluentValidation) on the pipeline structure before executing it.

[HttpPost("aggregate")]
public async Task<IActionResult> Aggregate([FromHeader] string custId, [FromBody] AggregateRequest request)
{
  
    var collection = db.GetCollection<BsonDocument>(request.CollectionName);

    // Validate/sanitize the pipeline here as needed
    var pipeline = BsonDocument.Parse(request.PipelineJson);

    var results = await collection.Aggregate<BsonDocument>(pipeline).ToListAsync();

    return Ok(results);
}
