using Xunit;
using Moq;
using FluentAssertions;
using MongoDB.Bson;
using MongoDB.Driver;
using MongoAPI.Services;
using System.Text.Json;

namespace MongoAPI.Tests.Services
{
    public class MongoDbServiceTests
    {
        private readonly Mock<IMongoClient> _mockClient;
        private readonly Mock<IMongoDatabase> _mockDatabase;
        private readonly Mock<IMongoCollection<BsonDocument>> _mockCollection;
        private readonly Mock<IConfiguration> _mockConfig;

        public MongoDbServiceTests()
        {
            _mockClient = new Mock<IMongoClient>();
            _mockDatabase = new Mock<IMongoDatabase>();
            _mockCollection = new Mock<IMongoCollection<BsonDocument>>();
            _mockConfig = new Mock<IConfiguration>();

            // Setup basic mocking chain
            _mockClient
                .Setup(c => c.GetDatabase(It.IsAny<string>(), null))
                .Returns(_mockDatabase.Object);

            _mockDatabase
                .Setup(d => d.GetCollection<BsonDocument>(It.IsAny<string>(), null))
                .Returns(_mockCollection.Object);

            _mockConfig
                .Setup(c => c.GetValue<string>("MongoDB:ConnectionString"))
                .Returns("mongodb://localhost:27017");
        }

        [Fact]
        public async Task FindOneAsync_ShouldReturnDocument_WhenDocumentExists()
        {
            // Arrange
            var service = new MongoDbService(_mockConfig.Object);
            var expectedDocument = new BsonDocument
            {
                { "_id", ObjectId.GenerateNewId() },
                { "name", "Test Document" }
            };

            var mockCursor = new Mock<IAsyncCursor<BsonDocument>>();
            mockCursor
                .Setup(c => c.MoveNextAsync(It.IsAny<CancellationToken>()))
                .ReturnsAsync(true);
            mockCursor
                .Setup(c => c.Current)
                .Returns(new[] { expectedDocument });

            _mockCollection
                .Setup(c => c.FindAsync(
                    It.IsAny<FilterDefinition<BsonDocument>>(),
                    It.IsAny<FindOptions<BsonDocument, BsonDocument>>(),
                    It.IsAny<CancellationToken>()))
                .ReturnsAsync(mockCursor.Object);

            // Act
            var result = await service.FindOneAsync(
                "testDb",
                "testCollection",
                new BsonDocument("name", "Test Document")
            );

            // Assert
            result.Should().NotBeNull();
            var resultDoc = BsonDocument.Parse(result.ToString());
            resultDoc["name"].AsString.Should().Be("Test Document");
        }

        [Fact]
        public async Task FindManyAsync_ShouldHandlePagination()
        {
            // Arrange
            var service = new MongoDbService(_mockConfig.Object);
            var documents = new List<BsonDocument>
            {
                new BsonDocument { { "name", "Doc1" } },
                new BsonDocument { { "name", "Doc2" } }
            };

            var mockCursor = new Mock<IAsyncCursor<BsonDocument>>();
            mockCursor
                .Setup(c => c.MoveNextAsync(It.IsAny<CancellationToken>()))
                .ReturnsAsync(true);
            mockCursor
                .Setup(c => c.Current)
                .Returns(documents);

            _mockCollection
                .Setup(c => c.FindAsync(
                    It.IsAny<FilterDefinition<BsonDocument>>(),
                    It.IsAny<FindOptions<BsonDocument, BsonDocument>>(),
                    It.IsAny<CancellationToken>()))
                .ReturnsAsync(mockCursor.Object);

            // Act
            var result = await service.FindManyAsync(
                "testDb",
                "testCollection",
                new BsonDocument(),
                null,
                new BsonDocument("name", 1),
                1,
                2
            );

            // Assert
            result.Should().NotBeNull();
            result.Length.Should().Be(2);
        }

        [Fact]
        public async Task InsertOneAsync_ShouldReturnId()
        {
            // Arrange
            var service = new MongoDbService(_mockConfig.Object);
            var document = new BsonDocument
            {
                { "_id", ObjectId.GenerateNewId() },
                { "name", "New Document" }
            };

            _mockCollection
                .Setup(c => c.InsertOneAsync(
                    It.IsAny<BsonDocument>(),
                    It.IsAny<InsertOneOptions>(),
                    It.IsAny<CancellationToken>()))
                .Returns(Task.CompletedTask);

            // Act
            var result = await service.InsertOneAsync(
                "testDb",
                "testCollection",
                document
            );

            // Assert
            result.Should().NotBeNull();
            ObjectId.TryParse(result, out _).Should().BeTrue();
        }

        [Fact]
        public async Task UpdateOneAsync_ShouldHandleUpsert()
        {
            // Arrange
            var service = new MongoDbService(_mockConfig.Object);
            var filter = new BsonDocument("_id", ObjectId.GenerateNewId());
            var update = new BsonDocument("$set", new BsonDocument("name", "Updated"));

            _mockCollection
                .Setup(c => c.UpdateOneAsync(
                    It.IsAny<FilterDefinition<BsonDocument>>(),
                    It.IsAny<UpdateDefinition<BsonDocument>>(),
                    It.IsAny<UpdateOptions>(),
                    It.IsAny<CancellationToken>()))
                .ReturnsAsync(new UpdateResult.Acknowledged(1, 1, BsonValue.Create(ObjectId.GenerateNewId())));

            // Act
            var result = await service.UpdateOneAsync(
                "testDb",
                "testCollection",
                filter,
                update,
                true
            );

            // Assert
            result.Should().NotBeNull();
            result.ModifiedCount.Should().Be(1);
            result.UpsertedId.Should().NotBeNull();
        }

        [Fact]
        public async Task DeleteOneAsync_ShouldReturnDeletedCount()
        {
            // Arrange
            var service = new MongoDbService(_mockConfig.Object);
            var filter = new BsonDocument("_id", ObjectId.GenerateNewId());

            _mockCollection
                .Setup(c => c.DeleteOneAsync(
                    It.IsAny<FilterDefinition<BsonDocument>>(),
                    It.IsAny<CancellationToken>()))
                .ReturnsAsync(new DeleteResult.Acknowledged(1));

            // Act
            var result = await service.DeleteOneAsync(
                "testDb",
                "testCollection",
                filter
            );

            // Assert
            result.Should().Be(1);
        }
    }
}
